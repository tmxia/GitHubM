2799| export async function getAllBranches(
2800|   owner: string,
2801|   repo: string
2802| ): Promise<GitHubBranch[]> {
2803|   const all: GitHubBranch[] = [];
2804|   const seen = new Set<string>();
2805|   for (let page = 1; page <= 20; page += 1) {
2806|     const result = await getBranches(owner, repo, page, 100);
2807|     for (const branch of result.data) {
2808|       if (seen.has(branch.name)) continue;
2809|       seen.add(branch.name);
2810|       all.push(branch);
2811|     }
2812|     if (!result.hasNextPage || result.data.length === 0) break;
2813|   }
2814|   return all;
2815| }
